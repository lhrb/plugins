cmake_minimum_required(VERSION 3.16)
project(nam_bridge LANGUAGES CXX)
set(CORE "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/NeuralAmpModelerCore")
# Match upstream tools/CMakeLists.txt without building its standalone tools.
file(GLOB NAM_SOURCES CONFIGURE_DEPENDS "${CORE}/NAM/*.cpp" "${CORE}/NAM/*/*.cpp")
add_library(nam_bridge STATIC nam_bridge.cpp ${NAM_SOURCES})
target_compile_features(nam_bridge PRIVATE cxx_std_20)
target_compile_definitions(nam_bridge PRIVATE NAM_SAMPLE_FLOAT NAM_ENABLE_A2_FAST)
target_include_directories(nam_bridge PRIVATE "${CORE}" "${CORE}/Dependencies/nlohmann")
target_include_directories(nam_bridge SYSTEM PRIVATE "${CORE}/Dependencies/eigen")
set_target_properties(nam_bridge PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden)
install(TARGETS nam_bridge ARCHIVE DESTINATION lib)

option(NAM_BRIDGE_TESTS "Build Linux allocation regression test" OFF)
if(NAM_BRIDGE_TESTS)
  enable_testing()
  add_executable(nam_process_test process_test.cpp)
  target_compile_features(nam_process_test PRIVATE cxx_std_20)
  target_link_libraries(nam_process_test PRIVATE "-Wl,--whole-archive" nam_bridge "-Wl,--no-whole-archive")
  target_link_options(nam_process_test PRIVATE "-Wl,--wrap=malloc,--wrap=calloc,--wrap=realloc")
  add_test(NAME nam_process_no_allocation COMMAND nam_process_test)
endif()
